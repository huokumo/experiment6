cmake_minimum_required(VERSION 3.16)
project(experiment6_navigation)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_FORTRESS_PLUGIN "Build the Gazebo Fortress planar teaching simulator" ON)
option(BUILD_UNITREE_ADAPTER "Build the optional R1 hardware adapter (requires unitree_sdk2)" OFF)
option(STANDALONE_PLUGIN "Build only the Fortress plugin, without ROS 2" OFF)

if(BUILD_FORTRESS_PLUGIN)
  find_package(ignition-gazebo6 REQUIRED)
  add_library(r1_planar_motion_system SHARED src/planar_motion_system.cpp)
  target_include_directories(r1_planar_motion_system PRIVATE include)
  target_link_libraries(r1_planar_motion_system ignition-gazebo6::ignition-gazebo6)
  target_compile_options(r1_planar_motion_system PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS r1_planar_motion_system LIBRARY DESTINATION lib)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(test_planar_geometry tests/test_planar_geometry.cpp)
  target_include_directories(test_planar_geometry PRIVATE include)
  add_test(NAME planar_geometry COMMAND test_planar_geometry)
  if(BUILD_FORTRESS_PLUGIN)
    add_executable(test_fortress_motion tests/test_fortress_motion.cpp)
    target_link_libraries(test_fortress_motion ignition-gazebo6::ignition-gazebo6)
    target_compile_definitions(test_fortress_motion PRIVATE
      PLUGIN_PATH="$<TARGET_FILE:r1_planar_motion_system>")
    add_dependencies(test_fortress_motion r1_planar_motion_system)
    add_test(NAME fortress_motion COMMAND test_fortress_motion)
    set_tests_properties(fortress_motion PROPERTIES TIMEOUT 60)
  endif()
endif()
if(STANDALONE_PLUGIN)
  return()
endif()

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(std_msgs REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/ObstacleSummary.msg" DEPENDENCIES std_msgs)
ament_python_install_package(experiment6_core)
ament_python_install_package(experiment6_student)
install(PROGRAMS scripts/pose_node.py scripts/motion_node.py scripts/depth_node.py
  scripts/drive_for.py scripts/avoidance_node.py scripts/student_avoidance_node.py
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config docs tools DESTINATION share/${PROJECT_NAME}
  PATTERN __pycache__ EXCLUDE)
install(FILES README.md LICENSE VALIDATION.md RELEASE_STATUS.json DESTINATION share/${PROJECT_NAME})

if(BUILD_UNITREE_ADAPTER)
  find_package(rclcpp REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(unitree_sdk2 REQUIRED)
  add_executable(r1_sdk_adapter src/r1_sdk_adapter.cpp)
  ament_target_dependencies(r1_sdk_adapter rclcpp geometry_msgs)
  target_link_libraries(r1_sdk_adapter unitree_sdk2)
  install(TARGETS r1_sdk_adapter DESTINATION lib/${PROJECT_NAME})
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
